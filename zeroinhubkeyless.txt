-- Tên file lưu cache 24h
local SAVE_FILE_NAME = "ZeroinHub_KeyCache.json"
local CACHE_DURATION = 86400 -- 24 giờ tính bằng giây (24 * 60 * 60)

-- Hàm kiểm tra xem đã lưu key và còn hạn hay không
local function checkSavedKey()
    if readfile and isfile and isfile(SAVE_FILE_NAME) then
        local success, data = pcall(function()
            return game:GetService("HttpService"):JSONDecode(readfile(SAVE_FILE_NAME))
        end)
        if success and data and data.key == CORRECT_KEY then
            local currentTime = os.time()
            if (currentTime - data.timestamp) < CACHE_DURATION then
                return true -- Key vẫn còn hiệu lực trong vòng 24h
            end
        end
    end
    return false
end

-- Hàm lưu key kèm thời gian
local function saveKeyCache()
    if writefile then
        local data = {
            key = CORRECT_KEY,
            timestamp = os.time()
        }
        pcall(function()
            writefile(SAVE_FILE_NAME, game:GetService("HttpService"):JSONEncode(data))
        end)
    end
end

-- Nếu đã lưu key hợp lệ trong 24h thì bỏ qua bảng nhập key và chạy thẳng script chính
if checkSavedKey() then
    local success, err = pcall(function()
        loadstring(game:HttpGet(SCRIPT_URL))()
    end)
    if not success then
        warn("Failed to load main script: " .. tostring(err))
    end
    return
end

-- Slow & Smooth Shake Animation
local function slowSmoothShake()
    local originalPos = UDim2.new(0.5, -210, 0.5, -135)
    local tweenInfo = TweenInfo.new(0.06, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut)
    
    task.spawn(function()
        TweenService:Create(MainFrame, tweenInfo, {Position = originalPos + UDim2.new(0, 10, 0, 0)}):Play()
        task.wait(0.06)
        TweenService:Create(MainFrame, tweenInfo, {Position = originalPos + UDim2.new(0, -10, 0, 0)}):Play()
        task.wait(0.06)
        TweenService:Create(MainFrame, tweenInfo, {Position = originalPos + UDim2.new(0, 6, 0, 0)}):Play()
        task.wait(0.06)
        TweenService:Create(MainFrame, tweenInfo, {Position = originalPos + UDim2.new(0, -6, 0, 0)}):Play()
        task.wait(0.06)
        TweenService:Create(MainFrame, tweenInfo, {Position = originalPos}):Play()
    end)
end

end)

-- Check Key Action
SubmitBtn.MouseButton1Click:Connect(function()
    local enteredKey = TextBox.Text
    
    if enteredKey == CORRECT_KEY then
        StatusLabel.TextColor3 = Color3.fromRGB(0, 255, 100)
        StatusLabel.Text = "Key verified! Loading script..."
        
        -- Lưu key vào cache để dùng trong 24h tới
        saveKeyCache()
        
        -- Smooth closing transition
        local tweenInfo = TweenInfo.new(0.4, Enum.EasingStyle.Quart, Enum.EasingDirection.Out)
        TweenService:Create(MainFrame, tweenInfo, {Size = UDim2.new(0, 0, 0, 0), Position = UDim2.new(0.5, 0, 0.5, 0)}):Play()
        
        task.wait(0.4)
        ScreenGui:Destroy()
        
        -- Run main script
        local success, err = pcall(function()
            loadstring(game:HttpGet(SCRIPT_URL))()
        end)
        
        if not success then
            warn("Failed to load main script: " .. tostring(err))
        end
    else
        StatusLabel.TextColor3 = Color3.fromRGB(255, 70, 70)
        StatusLabel.Text = "Invalid Key! Please try again."
        
        -- Xóa sạch chữ ngay lập tức và chạy hiệu ứng lắc chậm mượt
        TextBox.Text = ""
        slowSmoothShake()
    end
end)
